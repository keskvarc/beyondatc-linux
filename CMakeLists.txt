cmake_minimum_required(VERSION 3.16)
project(BeyondATCLinux VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(XPMP2_SOURCE_DIR "" CACHE PATH "Pinned XPMP2 source including the X-Plane SDK")
if(NOT EXISTS "${XPMP2_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Set XPMP2_SOURCE_DIR to XPMP2 fb543f9d063e7fc61e08f9d21c6e68136606fc23 (including submodules)")
endif()
add_subdirectory("${XPMP2_SOURCE_DIR}" xpmp2 EXCLUDE_FROM_ALL)
add_library(batc-native MODULE src/plugin.cpp)
target_link_libraries(batc-native PRIVATE XPMP2 pthread)
target_include_directories(batc-native PRIVATE
    "${XPMP2_SOURCE_DIR}/lib/SDK/CHeaders/XPLM"
    "${XPMP2_SOURCE_DIR}/lib/SDK/CHeaders/Widgets")
target_compile_options(batc-native PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
target_link_options(batc-native PRIVATE -static-libstdc++ -static-libgcc -Wl,--exclude-libs,ALL)
set_target_properties(batc-native PROPERTIES PREFIX "" OUTPUT_NAME lin SUFFIX ".xpl")

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/reference")
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/generate_fixtures.py"
        "${CMAKE_CURRENT_BINARY_DIR}/reference" RESULT_VARIABLE fixture_result)
    if(NOT fixture_result EQUAL 0)
        message(FATAL_ERROR "Packet fixture generation failed")
    endif()
    add_executable(wire-test tests/wire_test.cpp)
    add_executable(diagnostics-test tests/diagnostics_test.cpp)
    target_link_libraries(diagnostics-test PRIVATE pthread)
    add_executable(menu-test tests/menu_test.cpp)
    target_include_directories(menu-test PRIVATE "${XPMP2_SOURCE_DIR}/lib/SDK/CHeaders/XPLM")
    target_compile_definitions(menu-test PRIVATE LIN=1 XPLM210)
    foreach(target wire-test diagnostics-test menu-test)
        target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    endforeach()
    add_test(NAME wire-reference COMMAND wire-test "${CMAKE_CURRENT_BINARY_DIR}/reference")
    add_test(NAME plugin-menu COMMAND menu-test)
    add_test(NAME flight-diagnostics COMMAND diagnostics-test "${CMAKE_CURRENT_BINARY_DIR}/reference")
    add_test(NAME diagnostic-json COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_diagnostics.py"
        "${CMAKE_CURRENT_BINARY_DIR}/reference/native-diagnostics.jsonl")
    set_tests_properties(diagnostic-json PROPERTIES DEPENDS flight-diagnostics)
endif()

install(TARGETS batc-native LIBRARY DESTINATION lib/beyondatc-native/64)
install(DIRECTORY "${XPMP2_SOURCE_DIR}/Resources/" DESTINATION share/beyondatc-native/Resources)
install(FILES "${XPMP2_SOURCE_DIR}/LICENSE" DESTINATION share/beyondatc-native RENAME LICENSE.XPMP2)
install(FILES "${XPMP2_SOURCE_DIR}/lib/SDK/license.txt" DESTINATION share/beyondatc-native RENAME LICENSE.XPlaneSDK)
install(FILES LICENSE DESTINATION share/beyondatc-native)
